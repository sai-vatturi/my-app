import { Component, ElementRef, Input, OnInit, OnChanges, SimpleChanges } from '@angular/core';
import * as d3 from 'd3';

export interface ReleaseNode { id: string; label: string; type: 'system' | 'release' | 'env' | 'sub-release' | 'test'; children?: ReleaseNode[]; }

@Component({ selector: 'app-release-workflow', templateUrl: './release-workflow.component.html', styleUrls: ['./release-workflow.component.scss'] })
export class ReleaseWorkflowComponent implements OnInit, OnChanges {
  @Input() releaseData: ReleaseNode[];
  private zoom: any; private svg: any; private zoomGroup: any; isLocked = false;
  constructor(private el: ElementRef) {}

  ngOnInit() { if (this.releaseData) { this.renderChart(this.releaseData); } }
  ngOnChanges(changes: SimpleChanges) { if (changes['releaseData'] && this.releaseData) { this.renderChart(this.releaseData); } }

  renderChart(data: ReleaseNode[]) {
    const element = this.el.nativeElement.querySelector('#releaseFlow');
    if (!element) { console.warn('No element found with id #releaseFlow'); return; }

    d3.select(element).selectAll('svg').remove();
    const width = element.offsetWidth || 900, height = 400;
    this.svg = d3.select(element).append('svg').attr('width', width).attr('height', height);
    this.zoomGroup = this.svg.append('g').attr('class', 'zoom-group');

    const root = d3.hierarchy({ label: 'Root', children: data } as any).sort((a, b) => a.data.label.localeCompare(b.data.label));
    const treeLayout = d3.tree().size([height - 40, width - 160]);
    treeLayout(root);

    this.zoomGroup.selectAll('.link')
      .data(root.links())
      .enter().append('path')
      .attr('class', 'link')
      .attr('fill', 'none')
      .attr('stroke', '#ccc')
      .attr('stroke-width', 2)
      .attr('d', d3.linkHorizontal().x(d => d.y).y(d => d.x));

    const node = this.zoomGroup.selectAll('.node')
      .data(root.descendants())
      .enter().append('g')
      .attr('class', 'node')
      .attr('transform', d => `translate(${d.y},${d.x})`);

    node.append('rect')
      .attr('width', 130)
      .attr('height', 40)
      .attr('x', -65)
      .attr('y', -20)
      .attr('rx', 10)
      .attr('fill', d => this.getColor(d.data.type))
      .attr('stroke', '#888')
      .attr('stroke-width', 2);

    node.append('text')
      .attr('dy', 5)
      .attr('text-anchor', 'middle')
      .text(d => d.data.label)
      .style('font-size', '16px')
      .style('fill', '#222');

    this.zoom = d3.zoom()
      .scaleExtent([0.25, 2.5])
      .on('zoom', (event: any) => { if (!this.isLocked) { this.zoomGroup.attr('transform', event.transform); } });

    this.svg.call(this.zoom);
  }

  getColor(type: string): string {
    switch (type) {
      case 'system': return '#f5f5dc';
      case 'release': return '#e6e6fa';
      case 'env': return '#d0e6fa';
      case 'sub-release': return '#cfeecc';
      case 'test': return '#d9caff';
      default: return '#f5f5f5';
    }
  }

  zoomIn() { this.svg.transition().call(this.zoom.scaleBy, 1.2); }
  zoomOut() { this.svg.transition().call(this.zoom.scaleBy, 0.8); }
  fitToScreen() { this.svg.transition().call(this.zoom.transform, d3.zoomIdentity); }
  lockPanZoom() { this.isLocked = !this.isLocked; }
}

import { Component, OnInit } from '@angular/core';
import { FormBuilder, FormGroup } from '@angular/forms';

@Component({ selector: 'app-create-release', templateUrl: './create-release.component.html', styleUrls: ['./create-release.component.scss'] })
export class CreateReleaseComponent implements OnInit {
  releaseTypes = ['Major release', 'Hotfix', 'Data patch', 'Hotfix & Data patch'];
  squads = ['HK_SME', 'MY_SME', 'TH'];
  releaseForm: FormGroup;
  constructor(private fb: FormBuilder) {
    this.releaseForm = this.fb.group({
      releaseType: [''],
      squadsInvolved: [[]],
      changeRequestNumber: [''],
      jiraVersion: [''],
      plannedDate: [''],
      startDate: [''],
    });
  }
  ngOnInit(): void {}

  updateSquadSelection(event: any) {
    const checkbox = event.target;
    const squadsArray = this.releaseForm.get('squadsInvolved');
    if (checkbox.checked) {
      squadsArray.setValue([...squadsArray.value, checkbox.value]);
    } else {
      squadsArray.setValue(squadsArray.value.filter((value: string) => value !== checkbox.value));
    }
  }

  calculateStartDate() {
    const plannedDate = this.releaseForm.get('plannedDate').value;
    if (plannedDate) {
      const date = new Date(plannedDate);
      date.setDate(date.getDate() - 30);
      this.releaseForm.patchValue({ startDate: date.toISOString().split('T')[0] });
    }
  }

  onSubmit() {
    console.log(this.releaseForm.value);
  }
}
